local function applyEffects(player, character)
    local key = player.Name:lower()
    local config = OWNERS[key]
    if not config then return end

    local root = character:WaitForChild("HumanoidRootPart", 5)
    if not root then return end
    for _, n in ipairs({ ":3_Tag", ":3_Highlight" }) do
        local old = character:FindFirstChild(n)
        if old then old:Destroy() end
    end

    local billboard = Instance.new("BillboardGui")
    billboard.Name = ":3_Tag"
    billboard.AlwaysOnTop = true
    billboard.Size = UDim2.new(5, 0, 1.2, 0)
    billboard.StudsOffset = Vector3.new(0, 3.8, 0)
    billboard.Adornee = root

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = config.Text
    label.TextColor3 = config.TextColor
    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.TextStrokeTransparency = 0
    label.TextScaled = true
    label.Font = Enum.Font.SourceSansBold
    label.Parent = billboard
    billboard.Parent = character

    local highlight = Instance.new("Highlight")
    highlight.Name = ":3_Highlight"
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.OutlineTransparency = 0
    highlight.FillTransparency = 1   -- no fill: only the outer outline glows
    highlight.FillColor = config.TextColor
    highlight.OutlineColor = config.TextColor
    highlight.Adornee = character
    highlight.Parent = character

    local oldLight = root:FindFirstChild(":3_Light")
    if oldLight then oldLight:Destroy() end
    local light = Instance.new("PointLight")   -- soft glow cast outside the body
    light.Name = ":3_Light"
    light.Range = 9
    light.Brightness = 2
    light.Shadows = false
    light.Color = config.TextColor
    light.Parent = root

    fxObjects[key] = { hl = highlight, bb = billboard, light = light }
end

PATTERNS.Wings = function(i, N, t, r)
    local mitad = math.floor(N / 2)
    local izq = (i <= mitad)
    local idx = izq and i or (i - mitad)
    local total = izq and mitad or (N - mitad)
    local prog = (total > 1) and ((idx - 1) / (total - 1)) or 0.5
    local lado = izq and -1 or 1
    local aleteo = math.sin(t * 2.5) * 0.15
    local x = lado * (1.2 + prog * (r * 0.8))
    local y = 0.5 + math.sin(prog * math.pi * 0.8) * 2.2 + (aleteo * (prog + 0.2))
    local z = -0.3 + math.cos(prog * math.pi * 0.5) * 0.8
    local yaw = lado * (math.rad(30) + prog * math.rad(40))
    local pitch = math.rad(-10) + (aleteo * 0.2)
    local roll = -lado * (math.rad(15) + prog * math.rad(35))
    return Vector3.new(x, y, z), CFrame.Angles(pitch, yaw, roll)
end

local GOLD = 0.618034
local function TT() return os.clock() * math.max(math.abs(speed), 0.1) end

-- METEORS: every hat waits very high up (70-stud tower) above a RANDOM spot, falls accelerating
-- like a meteor, hits the ground and goes back to the sky to pick another random spot.
-- Spots are mostly FAR from you (25-80+ studs). Very small chance one lands on you,
-- and an even smaller chance it lands next to another player. Hats never collide with anyone.
local METEOR_CHANCE_USER = 0.04     -- 4%  lands on you
local METEOR_CHANCE_OTHER = 0.015   -- 1.5% lands next to another player
local meteorState = {}

local function groundAt(x, z, refY)
    local list = { refFolder }
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then table.insert(list, p.Character) end
    end
    local rp = RaycastParams.new()
    rp.FilterType = FILTER_EXCLUDE
    rp.FilterDescendantsInstances = list
    pcall(function() rp.RespectCanCollide = true end)
    local res = workspace:Raycast(Vector3.new(x, refY + 150, z), Vector3.new(0, -400, 0), rp)
    return res and res.Position.Y or (refY - 3)
end

PATTERNS.Meteors = function(i, N, t, r)
    local H, G = 70, 45
    local fall = math.sqrt(2 * H / G)
    local waitTop, pause = 1.2, 0.15
    local P = waitTop + fall + pause
    local clock = TT() + ((i * GOLD) % 1) * P
    local ph = clock % P
    local cycle = math.floor(clock / P)
    local userPos = targetHRP.Position

    -- choose a new landing spot every time this hat starts a new fall
    local ms = meteorState[i]
    if not ms or ms.cycle ~= cycle then
        local rng = Random.new(i * 7919 + cycle * 104729)
        local roll = rng:NextNumber()
        ms = { cycle = cycle }
        if roll < METEOR_CHANCE_USER then
            ms.track = "user"
            ms.ox, ms.oz = rng:NextNumber(-3, 3), rng:NextNumber(-3, 3)
        elseif roll < METEOR_CHANCE_USER + METEOR_CHANCE_OTHER then
            local others = {}
            for _, p in ipairs(Players:GetPlayers()) do
                local hr = (p ~= plr) and p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                if hr and (hr.Position - userPos).Magnitude < 150 then table.insert(others, p) end
            end
            if #others > 0 then
                ms.track = others[rng:NextInteger(1, #others)]
                local ang = rng:NextNumber(0, 2 * math.pi)
                ms.ox, ms.oz = math.cos(ang) * 4, math.sin(ang) * 4   -- lands beside them, not on top
            end
        end
        if not ms.track then
            local ang = rng:NextNumber(0, 2 * math.pi)
            local dist = rng:NextNumber(math.max(25, r * 3), math.max(80, r * 10))
            ms.x, ms.z = userPos.X + math.cos(ang) * dist, userPos.Z + math.sin(ang) * dist
            ms.gy = groundAt(ms.x, ms.z, userPos.Y)
        end
        meteorState[i] = ms
    end

    -- current landing spot (world space)
    local sx, sz, gy
    if ms.track == "user" then
        sx, sz, gy = userPos.X + ms.ox, userPos.Z + ms.oz, userPos.Y - 3
    elseif ms.track then
        local hr = ms.track.Character and ms.track.Character:FindFirstChild("HumanoidRootPart")
        if hr then
            sx, sz, gy = hr.Position.X + ms.ox, hr.Position.Z + ms.oz, hr.Position.Y - 3
        else
            sx, sz, gy = userPos.X + 40, userPos.Z, userPos.Y - 3
        end
    else
        sx, sz, gy = ms.x, ms.z, ms.gy
    end

    local landY = gy + 1
    local y, k = landY + H, 0
    if ph >= waitTop then
        local s = math.min(ph - waitTop, fall)
        y = math.max(landY + H - 0.5 * G * s * s, landY)
        k = s / fall
    end
    -- slight diagonal like a real meteor
    local drift = (1 - k) * 6
    return Vector3.new(sx - userPos.X + drift, y - userPos.Y, sz - userPos.Z), CFrame.Angles(math.rad(50), i, 0)
end

local function polyPoint(s, sides, rad)
    local f = s * sides
    local k = math.floor(f)
    local fr = f - k
    local a1 = k * 2 * math.pi / sides
    local a2 = (k + 1) * 2 * math.pi / sides
    local p1 = Vector3.new(math.cos(a1), 0, math.sin(a1))
    local p2 = Vector3.new(math.cos(a2), 0, math.sin(a2))
    return p1:Lerp(p2, fr) * rad
end

-- 3 firework shells: rise, burst into a sphere of hats, fall, repeat
PATTERNS.Fireworks = function(i, N, t, r)
    local g = (i - 1) % 3
    local P = 3.4
    local clock = TT() + g * P / 3
    local ph = clock % P
    local cyc = math.floor(clock / P)
    local a = hash(cyc * 1.7 + g * 11.3) * 2 * math.pi
    local rad = r * 0.9 * (0.4 + hash(cyc * 2.9 + g * 5.7) * 0.6)
    local bx, bz = math.cos(a) * rad, math.sin(a) * rad
    local rise = 1.3
    if ph < rise then
        local u = ph / rise
        return Vector3.new(bx, -2.5 + (1 - (1 - u) ^ 2) * 11, bz), CFrame.Angles(0, a, 0)
    end
    local s = ph - rise

local function polyhedron(verts, spinSpeed)
    local minD = math.huge
    for a = 1, #verts do
        for b = a + 1, #verts do
            minD = math.min(minD, (verts[a] - verts[b]).Magnitude)
        end
    end
    local edges = {}
    for a = 1, #verts do
        for b = a + 1, #verts do
            if math.abs((verts[a] - verts[b]).Magnitude - minD) < 1e-3 then table.insert(edges, { verts[a], verts[b] }) end
        end
    end
    local maxR = 0.001
    for _, v in ipairs(verts) do maxR = math.max(maxR, v.Magnitude) end
    local E = math.max(#edges, 1)
    return function(i, N, t, r)
        local e = edges[(i - 1) % E + 1]
        local layers = math.max(math.ceil(N / E), 1)
        local f = (math.floor((i - 1) / E) + 0.5) / layers
        local g = CFrame.Angles(t * 0.5 * spinSpeed, t * 0.7 * spinSpeed, t * 0.2)
        return g * (e[1]:Lerp(e[2], f) * (r * 0.8 / maxR)) + V(0, 1.5, 0), g
    end
end

local HILB = {}
for d = 0, 63 do
    local tt, x, y, s = d, 0, 0, 1
    while s < 8 do
        local rx = math.floor(tt / 2) % 2
        local ry = ((tt % 2) + rx) % 2
        if ry == 0 then
            if rx == 1 then x = s - 1 - x; y = s - 1 - y end
            x, y = y, x
        end
        x = x + s * rx
        y = y + s * ry
        tt = math.floor(tt / 4)
        s = s * 2
    end
    HILB[d + 1] = Vector2.new(x, y)
end

local function lobes(count, vertical)
    return function(i, N, t, r)
        local L = (i - 1) % count
        local k = math.floor((i - 1) / count)
        local per = math.max(math.ceil(N / count), 1)
        local center
        if vertical then
            center = V(0, 1.5 + ((L == 0) and 1 or -1) * r * 0.5, 0)
        else
            local ca = L * TAU / count + t * 0.3
            center = V(math.cos(ca) * r * 0.55, 1, math.sin(ca) * r * 0.55)
        end
        local sa = t * 3 + k * TAU / per

        return center + V(math.cos(sa) * lr, math.sin(sa * 1.3) * 0.5, math.sin(sa) * lr), CFrame.Angles(0, -sa, 0)
    end
end

    if p > 0.20 and p < 0.26 then return -0.2 * math.sin((p - 0.20) / 0.06 * math.pi)
    elseif p >= 0.26 and p < 0.32 then return 1.8 * math.sin((p - 0.26) / 0.06 * math.pi)
    elseif p >= 0.32 and p < 0.38 then return -0.5 * math.sin((p - 0.32) / 0.06 * math.pi)
    elseif p > 0.55 and p < 0.75 then return 0.35 * math.sin((p - 0.55) / 0.2 * math.pi)
    end
    return 0
end

local function stadium(s, S, R)
    local L = 2 * S + 2 * math.pi * R
    local d = (s % 1) * L
    if d < S then return -S / 2 + d, -R end
    d = d - S
    if d < math.pi * R then
        local ph = -math.pi / 2 + d / R
        return S / 2 + math.cos(ph) * R, math.sin(ph) * R
    end
    d = d - math.pi * R
    if d < S then return S / 2 - d, R end
    d = d - S
    local ph = math.pi / 2 + d / R
    return -S / 2 + math.cos(ph) * R, math.sin(ph) * R
end

-- ===================== NPC AI (they walk on the ground, no chat) =====================
-- social = true  -> they approach other players, look at them and greet them with a gesture
-- social = false -> they just wander around
local frameRP = nil -- raycast params computed once per frame (performs well with 12+ accessories)

-- Half of the accessory's real height (detects large size, mesh scale and mesh offset)
local function halfHeight(h)
    if not h or not h.Parent then return 0.8 end
    local y = h.Size.Y
    local off = 0
    local sm = h:FindFirstChildOfClass("SpecialMesh")
    if sm then
        y = math.max(y, y * sm.Scale.Y)
        off = sm.Offset.Y
    end
    return math.max((y / 2 - off) * 1.08, 0.3)
end

local function floorParams()
    local lista = { refFolder }
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then table.insert(lista, p.Character) end
    end
    local rp = RaycastParams.new()
    rp.FilterType = FILTER_EXCLUDE
    rp.FilterDescendantsInstances = lista
    pcall(function() rp.RespectCanCollide = true end)
    return rp
end

-- ===================== RELAY (NPC <-> NPC between script users) =====================
-- Only canned lines (by index) travel over the relay, never free text. Chat lines are
-- addressed to ONE user, so only the two users involved ever see them (and only inside the script UI).
local Debris = game:GetService("Debris")
local npcIdCounter = 0
local relay = { ws = nil, connected = false, status = (RELAY_URL == "") and "Relay: not configured (NPC chat is local-only)" or "Relay: connecting..." }
local remote = {}                 -- remote[uid] = { name, seen, npcs = { [id] = { label, pos } } }
local pushChat = function() end   -- replaced by the chat panel in createGUI

local LINES = {
    [1] = { "Hey there!", "Hi! Nice hat!", "Hello, fellow NPC!" },
    [2] = { "Oh, hello!", "Hey! Didn't see you there.", "Hi! Beautiful day, isn't it?" },
    [3] = { "What are you up to?", "Wanna wander around together?", "Seen anything interesting?" },
    [4] = { "Just walking around.", "Sure, why not!", "Not much, just vibing." },
    [5] = { "Well, see you later!", "Bye bye!", "Catch you around!" },
    [6] = { "Bye!", "Take care!", "See ya!" },
}

local function relaySend(tbl)
    if not relay.ws or not relay.connected then return end
    local ok, js = pcall(function() return HttpService:JSONEncode(tbl) end)
    if ok then pcall(function() relay.ws:Send(js) end) end
end

local function cleanNum(n)
    if type(n) == "number" and n == n and math.abs(n) < 1e6 then return n end
    return nil
end

local function npcStateById(id)
    for _, st in pairs(npcStates) do
        if st.id == id then return st end
    end
    return nil
end

-- speech bubble above MY npc (local only, nobody else sees it)
local function showBubble(st, text)
    local h = st.handle
    if not h or not h.Parent then return end
    local old = h:FindFirstChild("NpcBubble")
    if old then old:Destroy() end
    local gui = Instance.new("BillboardGui")
    gui.Name = "NpcBubble"
    gui.Adornee = h
    gui.AlwaysOnTop = true
    gui.Size = UDim2.new(0, 170, 0, 34)
    gui.StudsOffset = Vector3.new(0, 3, 0)
    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, 0, 1, 0)
    lbl.BackgroundColor3 = Color3.new(1, 1, 1)
    lbl.BackgroundTransparency = 0.1
    lbl.TextColor3 = Color3.new(0, 0, 0)
    lbl.Font = Enum.Font.GothamMedium
    lbl.TextSize = 13
    lbl.TextWrapped = true
    lbl.Text = text
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 8)
    c.Parent = lbl
    lbl.Parent = gui
    gui.Parent = h
    Debris:AddItem(gui, 3.5)
end

local function speakNpc(st, uid, fid, step, v)
    relaySend({ t = "c", u = plr.UserId, n = plr.Name, to = uid, fid = st.id, tid = fid, step = step, v = v, fl = st.label })
    local text = LINES[step][v]
    pushChat(plr.Name .. "'s " .. st.label, text, Color3.fromRGB(120, 200, 255))
    showBubble(st, text)
end

local function onNpcChat(uid, m)
    local step, v = tonumber(m.step), tonumber(m.v)
    if not step or not v or step < 1 or step > 6 then return end
    if not LINES[step][v] then return end
    local st = npcStateById(tonumber(m.tid))
    if not st then return end
    local fid = tonumber(m.fid)
    if not fid then return end
    local now = os.clock()
    local r = remote[uid]
    local fromName = tostring(m.n or (r and r.name) or "?"):sub(1, 24)
    local fromLabel = tostring(m.fl or "NPC"):sub(1, 20)
    pushChat(fromName .. "'s " .. fromLabel, LINES[step][v], Color3.fromRGB(255, 220, 120))

    if st.state ~= "visit" then
        st.talkTo = { uid = uid, id = fid }
        st.talkUntil = now + 8
    elseif step >= 6 then
        st.visitEnd = math.min(st.visitEnd, now + 2.5)
    end

    if step < 6 then
        local nextStep = step + 1
        task.delay(1.6 + math.random() * 1.4, function()
            if not getgenv().OrbitToolsy then return end
            if npcStateById(st.id) ~= st then return end
            speakNpc(st, uid, fid, nextStep, math.random(#LINES[nextStep]))
            if nextStep >= 6 and st.state ~= "visit" then st.talkUntil = os.clock() + 3 end
        end)
    end
end

local function onRelayMessage(raw)
    local ok, m = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok or type(m) ~= "table" then return end
    local uid = tonumber(m.u)
    if not uid or uid == plr.UserId then return end
    if m.t == "s" then
        local r = remote[uid] or { npcs = {} }
        r.name = tostring(m.n or "?"):sub(1, 24)
        r.seen = os.clock()
        r.npcs = {}
        if type(m.p) == "table" then
            for k, e in ipairs(m.p) do
                if k > 40 then break end
                if type(e) == "table" then
                    local id, x, y, z = tonumber(e.id), cleanNum(e.x), cleanNum(e.y), cleanNum(e.z)
                    if id and x and y and z then
                        r.npcs[id] = { label = tostring(e.l or "NPC"):sub(1, 20), pos = Vector3.new(x, y, z) }
                    end
                end
            end
        end
        remote[uid] = r
    elseif m.t == "c" and tonumber(m.to) == plr.UserId then
        onNpcChat(uid, m)
    end
end

local function connectRelay()
    if RELAY_URL == "" then return end
    local conn = (WebSocket and WebSocket.connect) or (syn and syn.websocket and syn.websocket.connect)
    if not conn then relay.status = "Relay: this executor has no WebSocket"; return end
    local ok, ws = pcall(conn, RELAY_URL)
    if not ok or not ws then relay.status = "Relay: connection failed (retrying)"; return end
    relay.ws = ws
    relay.connected = true
    relay.status = "Relay: connected"
    pcall(function() ws.OnMessage:Connect(onRelayMessage) end)
    pcall(function()
        ws.OnClose:Connect(function()
            relay.connected = false
            relay.ws = nil
            relay.status = "Relay: disconnected (retrying)"
        end)
    end)
    local room = (game.JobId ~= "") and game.JobId or tostring(game.PlaceId)
    relaySend({ t = "join", room = room, u = plr.UserId, n = plr.Name })
end

-- ===================== NPC GESTURES / WALKER =====================
local GESTOS = {
    wave = 1.2, spin = 1.0, bow = 1.4, dance = 1.8,
}
local GESTOS_LISTA = { "wave", "spin", "bow", "dance" }

local function flat(v) return Vector3.new(v.X, 0, v.Z) end

local function pickDest(center, range)
    local a = math.random() * 2 * math.pi
    local d = math.random(6, math.max(8, math.floor(range)))
    return center + Vector3.new(math.cos(a) * d, 0, math.sin(a) * d)
end

-- Picks who to visit: ignores some players (randomly) and recently visited ones
local function pickVisitTarget(st, now)
    local cands = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r and (r.Position - st.pos).Magnitude < 70 then
                local id = p.UserId
                if st.ignore[id] == nil then
                    -- 35% chance to ignore this player for 1-2.5 minutes
                    st.ignore[id] = (math.random() < 0.35) and (now + math.random(60, 150)) or 0
                end
                if st.ignore[id] <= now and (st.cool[id] or 0) <= now then
                    table.insert(cands, p)
                end
            end
        end
    end
    if #cands == 0 then return nil end
    return cands[math.random(#cands)]
end

-- Picks an NPC of ANOTHER script user (needs the relay)
local function pickRemoteNpc(st, now)
    local cands = {}
    for uid, r in pairs(remote) do
        if now - r.seen < 4 then
            for id, e in pairs(r.npcs) do
                if (e.pos - st.pos).Magnitude < 70 then
                    local key = "n" .. uid .. "_" .. id
                    if st.ignore[key] == nil then
                        st.ignore[key] = (math.random() < 0.3) and (now + math.random(40, 100)) or 0
                    end
                    if st.ignore[key] <= now and (st.cool[key] or 0) <= now then
                        table.insert(cands, { uid = uid, id = id, label = e.label, key = key })
                    end
                end
            end
        end
    end
    if #cands == 0 then return nil end
    return cands[math.random(#cands)]
end

local FOLLOW_MAX = 14          -- follow-owner mode: farthest an NPC may be from you before it runs back
local NPC_PATTERNS = { NPC = true, NPCSocial = true, NPCScared = true }

-- key = reference part of this hat (unique state key)
-- social = approaches/greets people | scared = runs away from other players
local function npcWalker(i, dt, key, handle, root, social, half, scared)
    local now = os.clock()
    local st = npcStates[key]
    if not st then
        local hp = (handle and handle.Parent) and handle.Position or key.Position
        local start = Vector3.new(hp.X + math.random(-3, 3), hp.Y, hp.Z + math.random(-3, 3))
        npcIdCounter = npcIdCounter + 1
        st = {
            id = npcIdCounter,
            label = tostring((handle and handle.Parent and handle.Parent.Name) or "NPC"):sub(1, 20),
            pos = start, home = start, dest = nil,
            state = "idle", timer = math.random() * 1.5,   -- they start moving almost immediately
            cara = CFrame.Angles(0, math.random() * 6.28, 0).LookVector,
            paso = 0, speedMul = 0.85 + math.random() * 0.4, fleeing = false,
            target = nil, tnpc = nil, phase = nil, visitEnd = 0, approachStart = 0,
            convoAt = 0, convoDone = false, talkTo = nil, talkUntil = 0,
            ignore = {}, cool = {},
            gesture = nil, gStart = 0, gNext = 0,
        }
        npcStates[key] = st
    end
    st.handle = handle
    st.paso = st.paso + dt

    local PARAR = 6
    local following = npcFollowOwner
    local center = following and root.Position or st.home
    local wanderRange = following and math.min(npcRange, 10) or npcRange
    local moveDir, lookDir = nil, nil
    local speedNow = 1

    local function endVisit()
        if st.tnpc then st.cool[st.tnpc.key] = now + math.random(30, 60) end
        if st.target then
            -- the owner gets visited again soon; other players only after a long break
            local cd = (st.target == plr) and math.random(6, 12) or math.random(25, 50)
            st.cool[st.target.UserId] = now + cd
        end
        st.target = nil
        st.tnpc = nil
        st.convoDone = false
        st.phase = nil
        if math.random() < 0.7 then
            st.state = "walk"
            st.dest = pickDest(center, wanderRange)
        else
            st.state = "idle"
            st.timer = math.random(2, 5)
        end
    end

    local function tryGesture()
        if not st.gesture and now >= st.gNext then
            st.gesture = GESTOS_LISTA[math.random(#GESTOS_LISTA)]
            st.gStart = now
            st.gNext = now + GESTOS[st.gesture] + math.random(2, 4)
        end
    end

    -- scared NPCs: nearest other player within 30 studs is a threat
    local threatPos
    if scared then
        local best = 30
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= plr and p.Character then
                local r = p.Character:FindFirstChild("HumanoidRootPart")
                if r then
                    local d = flat(r.Position - st.pos).Magnitude
                    if d < best then best = d; threatPos = r.Position end
                end
            end
        end
    end

    local talking = st.talkUntil and now < st.talkUntil and st.state ~= "visit"
    st.fleeing = false

    if threatPos then
        -- RUN AWAY (zig-zag), but never too far from the area they belong to
        st.fleeing = true
        local away = flat(st.pos - threatPos)
        if away.Magnitude < 0.1 then away = Vector3.new(math.random() - 0.5, 0, math.random() - 0.5) end
        local dir = CFrame.Angles(0, math.sin(now * 6 + st.id) * 0.5, 0) * away.Unit
        local leashR = following and (FOLLOW_MAX * 1.6) or (npcRange * 2.5)
        local toC = flat(center - st.pos)
        if toC.Magnitude > leashR then dir = (dir + toC.Unit * 1.5).Unit end
        moveDir = dir * 100
        lookDir = dir
        speedNow = 2.3
    elseif talking then
        -- another script user's NPC is talking to this one: stand still, face it, gesture
        local rr = st.talkTo and remote[st.talkTo.uid]
        local e = rr and rr.npcs[st.talkTo.id]
        if e then lookDir = flat(e.pos - st.pos) end
        tryGesture()
    else
        -- FOLLOW-OWNER MODE (strict): stay close to you; if too far, run back to you
        if following then
            if st.state == "visit" and st.target ~= plr then endVisit() end   -- only you matter in this mode
            if st.state ~= "visit" and st.state ~= "return" and flat(root.Position - st.pos).Magnitude > FOLLOW_MAX then
                st.state = "return"
            end
        end

        if st.state == "return" then
            local f = flat(root.Position - st.pos)
            if f.Magnitude < 8 then
                st.state = "idle"
                st.timer = math.random(1, 2)
            else
                moveDir = f
                speedNow = math.clamp(f.Magnitude / 14, 1.5, 4)   -- the farther you are, the faster they run
            end
        elseif st.state == "idle" then
            st.timer = st.timer - dt
            if st.timer <= 0 then
                local chosen, chosenNpc = nil, nil
                if social then
                    if following then
                        -- go straight to the owner and interact
                        if (st.cool[plr.UserId] or 0) <= now and math.random() < 0.65 then chosen = plr end
                    elseif math.random() < 0.5 then
                        if relay.connected and math.random() < 0.5 then chosenNpc = pickRemoteNpc(st, now) end
                        if not chosenNpc then chosen = pickVisitTarget(st, now) end
                    end
                end
                if chosenNpc then
                    st.state = "visit"
                    st.tnpc = chosenNpc
                    st.target = nil
                    st.phase = "approach"
                    st.approachStart = now
                elseif chosen then
                    st.state = "visit"
                    st.target = chosen
                    st.tnpc = nil
                    st.phase = "approach"
                    st.approachStart = now
                else
                    st.state = "walk"
                    st.dest = pickDest(center, wanderRange)
                end
            end
        elseif st.state == "walk" then
            local f = flat(st.dest - st.pos)
            if f.Magnitude < 0.6 then
                st.state = "idle"
                st.timer = math.random(1, 3)
            else
                moveDir = f
            end
        elseif st.state == "visit" then
            local tpos
            if st.tnpc then
                local rr = remote[st.tnpc.uid]
                local e = rr and (now - rr.seen < 4) and rr.npcs[st.tnpc.id]
                tpos = e and e.pos or nil
            else
                local ch = st.target and st.target.Character
                local r = ch and ch:FindFirstChild("HumanoidRootPart")
                tpos = r and r.Position or nil
            end
            if not tpos then
                endVisit()
            else
                local f = flat(tpos - st.pos)
                lookDir = f
                if st.phase == "approach" then
                    if f.Magnitude > PARAR + 0.5 then
                        moveDir = f
                        if st.target == plr then speedNow = math.clamp(f.Magnitude / 14, 1, 3.5) end -- keep up with you
                        if now - st.approachStart > 20 then endVisit() end -- couldn't reach them: gives up
                    else
                        st.phase = "interact"
                        st.visitEnd = now + (st.tnpc and 26 or math.random(4, 9))
                        st.gNext = now
                        st.convoAt = now + 0.6
                        st.convoDone = false
                    end
                else
                    if f.Magnitude > PARAR + 4 then
                        st.phase = "approach"
                        st.approachStart = now
                    end
                    tryGesture()
                    if st.tnpc and not st.convoDone and now >= st.convoAt then
                        st.convoDone = true
                        speakNpc(st, st.tnpc.uid, st.tnpc.id, 1, math.random(#LINES[1]))
                    end
                    if now >= st.visitEnd then endVisit() end
                end
            end
        end
    end

    local moviendo = false
    if moveDir and moveDir.Magnitude > 0.1 then
        moviendo = true
        local sp = npcSpeed * st.speedMul * speedNow
        st.pos = st.pos + moveDir.Unit * math.min(sp * dt, moveDir.Magnitude)
        lookDir = lookDir or moveDir
    end

    -- separation: NPCs don't pile up on each other (helps them spread out)
    for k, o in pairs(npcStates) do
        if o ~= st then
            local d = flat(st.pos - o.pos)
            local m = d.Magnitude
            if m < 2.5 then
                local dir = (m > 0.01) and d.Unit or CFrame.Angles(0, math.random() * 6.28, 0).LookVector
                st.pos = st.pos + dir * (2.5 - m) * math.min(dt * 3, 1)
            end
        end
    end

    if lookDir and lookDir.Magnitude > 0.05 then
        st.cara = st.cara:Lerp(lookDir.Unit, math.clamp(dt * 8, 0, 1))
        if st.cara.Magnitude < 0.01 then st.cara = lookDir.Unit end
    end

    -- gestures (movement only)
    local salto, ladeo, spin, cabeceo = 0, 0, 0, 0
    if st.gesture then
        local prog = (now - st.gStart) / GESTOS[st.gesture]
        if prog >= 1 then
            st.gesture = nil
        elseif st.gesture == "wave" then
            salto = math.abs(math.sin(prog * math.pi * 2)) * 0.8
            ladeo = math.sin(prog * math.pi * 6) * math.rad(15)
        elseif st.gesture == "spin" then
            spin = prog * 2 * math.pi
            salto = math.sin(prog * math.pi) * 0.6
        elseif st.gesture == "bow" then
            cabeceo = math.sin(prog * math.pi) * math.rad(30)
        elseif st.gesture == "dance" then
            ladeo = math.sin(prog * math.pi * 8) * math.rad(20)
            salto = math.abs(math.sin(prog * math.pi * 4)) * 0.5
        end
    end

    -- ground
    local res = workspace:Raycast(st.pos + Vector3.new(0, 10, 0), Vector3.new(0, -60, 0), frameRP or floorParams())
    local sueloY = res and res.Position.Y or st.pos.Y
    st.pos = Vector3.new(st.pos.X, sueloY, st.pos.Z)

    -- fleeing NPCs hop faster and shake in panic
    local rate = st.fleeing and 16 or 10
    local rebote = moviendo and math.abs(math.sin(st.paso * rate)) * (st.fleeing and 0.5 or 0.35) or 0
    local inclin = moviendo and math.sin(st.paso * rate) * math.rad(st.fleeing and 14 or 6) or 0
    local yaw = math.atan2(-st.cara.X, -st.cara.Z)

    -- when tilted, a large accessory needs more height so it doesn't sink into the ground
    local tilt = math.max(math.abs(cabeceo), math.abs(ladeo + inclin))
    local lift = (half or 0.8) / math.max(math.cos(tilt), 0.7)

    return CFrame.new(st.pos + Vector3.new(0, lift + 0.1 + rebote + salto, 0))
        * CFrame.Angles(0, yaw + spin, 0) * CFrame.Angles(cabeceo, 0, inclin + ladeo)
end

local modeListeners = {}
local function setMode(n)
    n = math.floor(tonumber(n) or mode)
    if n < 1 then n = 1 end
    if n > #AURAS then n = #AURAS end

-- ===================== ITEM LOGIC =====================
local function waitForHRP()
    while not hrp or not hrp.Parent do
        hrp = chr and chr:FindFirstChild("HumanoidRootPart")
        if not hrp then task.wait() end
    end
end

local cleanupTool

local function setupOrbitingItem(item, handle)
    if not handle or not handle.Parent then return end
    if table.find(handles, handle) then
        cleanupTool(handle)
        task.wait()
    end
    waitForHRP()

    if item:IsA("Accessory") then
        handle.Massless = true
        handle.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0, 0, 0)
        handle.CanCollide = false
        handle:BreakJoints()
        for _, w in ipairs(handle:GetChildren()) do
            if w:IsA("Weld") or w:IsA("Motor6D") then w:Destroy() end
        end
        handle.AssemblyLinearVelocity = Vector3.zero
        handle.AssemblyAngularVelocity = Vector3.zero
    end

    if item:IsA("Tool") then
        local arm = chr:FindFirstChild("Right Arm") or chr:FindFirstChild("RightHand")
        if arm then
            local grip = arm:FindFirstChild("RightGrip")
            if grip then grip:Destroy() end
        end
        if item.Parent ~= chr then item.Parent = chr end
    end

    connections[handle] = item.AncestryChanged:Connect(function(_, parent)
        if parent ~= chr then cleanupTool(handle) end
    end)

    table.insert(handles, handle)
    table.insert(toolNames, item.Name)
    local index = #handles
    lastResyncTime[handle] = 0

    local p = Instance.new("Part")
    p.Name = "OrbitReference_" .. item.Name
    p.Anchored = true
    p.CanCollide = false
    p.CanQuery = false
    p.CanTouch = false
    p.Transparency = 1
    p.Size = Vector3.new(0.2, 0.2, 0.2)
    p.Position = hrp.Position
    p.Parent = refFolder
    orbitParts[index] = p

    local a0 = Instance.new("Attachment", handle)
    local a1 = Instance.new("Attachment", p)

    local ap = Instance.new("AlignPosition")
    ap.MaxForce = math.huge
    ap.MaxVelocity = math.huge
    ap.Responsiveness = 200
    ap.Attachment0 = a0
    ap.Attachment1 = a1
    ap.Parent = handle

    local ao = Instance.new("AlignOrientation")
    ao.Mode = Enum.OrientationAlignmentMode.TwoAttachment
    ao.MaxTorque = math.huge
    ao.MaxAngularVelocity = math.huge
    ao.Responsiveness = 200
    ao.RigidityEnabled = false
    ao.Attachment0 = a0
    ao.Attachment1 = a1
    ao.Parent = handle

    movers[handle] = { align = ap, orient = ao, a0 = a0 }
    handle.CFrame = p.CFrame

    task.defer(function()
        if handle and handle.Parent then handle.CFrame = p.CFrame end
        task.wait(0.05)
        if handle and handle.Parent then handle.CFrame = p.CFrame end
    end)
end

local function setupTool(v)
    if not toolsEnabled then return end
    local h = v:FindFirstChild("Handle")
    if h then setupOrbitingItem(v, h) end
end

local function setupAccessory(v)
    if getgenv().OrbitAccBlacklist[v.Name] then return end
    local h = v:FindFirstChild("Handle")
    if h then setupOrbitingItem(v, h) end
end

cleanupTool = function(h)
    local index = table.find(handles, h)
    if not index then return end
    lastResyncTime[h] = nil
    if orbitParts[index] then
        npcStates[orbitParts[index]] = nil
        anchorSnap[orbitParts[index]] = nil
        orbitParts[index]:Destroy()
    end
    local m = movers[h]
    if m then
        if m.align then m.align:Destroy() end
        if m.orient then m.orient:Destroy() end
        if m.a0 then m.a0:Destroy() end
        movers[h] = nil
    end
    if connections[h] then
        connections[h]:Disconnect()
        connections[h] = nil
    end
    table.remove(orbitParts, index)
    table.remove(handles, index)
    table.remove(toolNames, index)
end

local function removeAllTools()
    for i = #handles, 1, -1 do
        local h = handles[i]
        local parent = h and h.Parent
        if parent and parent:IsA("Tool") then cleanupTool(h) end
    end
end

local function addAllExistingTools()
    if not toolsEnabled then return end
    for _, v in ipairs(chr:GetChildren()) do
        if v:IsA("Tool") then setupTool(v) end
    end
end

local function setupTargetMonitor(root)
    if targetMonitor then targetMonitor:Disconnect(); targetMonitor = nil end
    if not root then return end
    local c = root.Parent
    if c and c:IsA("Model") then
        targetMonitor = c.AncestryChanged:Connect(function()
            if not c.Parent or c.Parent ~= workspace then
                targetHRP = hrp
                if targetMonitor then targetMonitor:Disconnect() end
                targetMonitor = nil
            end
        end)
    end
end

table.insert(conns, RunService.RenderStepped:Connect(function(dt)
    if not getgenv().OrbitToolsy then return end
    if not targetHRP or not targetHRP.Parent then
        targetHRP = hrp
        if targetMonitor then targetMonitor:Disconnect(); targetMonitor = nil end
    end

    local num = #orbitParts
    if num == 0 then return end

    rot = rot + speed
    local time = os.clock()
    if mode ~= ANCHOR_MODE then table.clear(anchorSnap) end

    local def = AURAS[mode]
    local pattern = def.Pattern
    local npcMode = NPC_PATTERNS[pattern] == true
    frameRP = npcMode and floorParams() or nil
    if not npcMode then table.clear(npcStates) end -- when returning to an NPC aura they start next to you

    for i, p in ipairs(orbitParts) do
        if p and p.Parent then
            local target
            local esNPC = false

            if NPC_PATTERNS[pattern] then
                esNPC = true
                target = npcWalker(i, dt, p, handles[i], targetHRP, pattern == "NPCSocial", halfHeight(handles[i]), pattern == "NPCScared")
                    + Vector3.new(sideOffset, heightOffset, -forwardOffset)
            elseif pattern == "Anchor" then
                if not anchorSnap[p] then anchorSnap[p] = p.CFrame end
                target = anchorSnap[p]
            else
                local fn = PATTERNS[pattern] or PATTERNS.Ring
                local pos, cfRot = fn(i, num, math.rad(rot), offset)
                local base = REALTIME[pattern] and CFrame.new(targetHRP.Position) or targetHRP.CFrame
                target = base * CFrame.new(sideOffset, heightOffset, -forwardOffset) * CFrame.new(pos) * cfRot
            end

            -- "look at user" button (does not affect NPCs, which look according to their AI)
            if lookAtUser and not esNPC and targetHRP and targetHRP.Parent then
                local pos = target.Position
                if (targetHRP.Position - pos).Magnitude > 0.1 then
                    target = CFrame.lookAt(pos, targetHRP.Position)
                end
            end

            if toolRotSpeed ~= 0 and not esNPC then
                target = target * CFrame.Angles(0, time * toolRotSpeed * 3, 0)
            end

            -- NPCs need to follow their destination quickly (with low smoothness they looked frozen)
            local rapido = REALTIME[pattern]
            local ls = esNPC and math.max(lerpSpeed, 8) or (rapido and math.max(lerpSpeed, 25) or lerpSpeed)
            local alpha = 1 - math.exp(-ls * dt)
            if rapido and (target.Position - p.Position).Magnitude > 12 then
                p.CFrame = target -- the meteor returns to the sky instantly (no slow trip)
            else
                p.CFrame = p.CFrame:Lerp(target, alpha)
            end
        end
    end

    if sethiddenproperty then
        pcall(function() sethiddenproperty(plr, "SimulationRadius", math.huge) end)
    end
end))

table.insert(conns, RunService.PostSimulation:Connect(function(dt)
    if not targetHRP or not targetHRP.Parent or not hrp or not hrp.Parent then return end
    local now = os.clock()
    local cur = hrp.Position
    if dt > 0 then hrpVel = (cur - lastHRPPos) / dt end
    lastHRPPos = cur
    local predicted = cur + hrpVel * 0.1
    local antiSleep = Vector3.new(0, math.sin(now * 15) * 0.0015, 0)
    local gravityAxis = SETTINGS.VelocityY + math.sin(now)
    local pa = AURAS[mode].Pattern
    local npcMode = (NPC_PATTERNS[pa] == true) or (REALTIME[pa] == true)
    for _, h in ipairs(handles) do
        if h and h:IsA("BasePart") then
            local dir = predicted - h.Position
            local xz = Vector3.new(dir.X, 0, dir.Z)
            local vxz = Vector3.zero
            if xz.Magnitude > 0 and not npcMode then vxz = xz.Unit * xz.Magnitude * 2 end
            h.AssemblyLinearVelocity = Vector3.new(vxz.X, gravityAxis, vxz.Z)
            h.AssemblyAngularVelocity = Vector3.zero
            h.CFrame = h.CFrame + antiSleep
        end
    end
end))

-- resync items that drift too far
task.spawn(function()
    while getgenv().OrbitToolsy do
        task.wait(0.1)
        local now = os.clock()
        for i = #handles, 1, -1 do
            local h = handles[i]
            local op = orbitParts[i]
            if h and h.Parent and op then
                local pa = AURAS[mode].Pattern
                local maxDist = (NPC_PATTERNS[pa] or REALTIME[pa]) and 400 or MAX_ORBIT_DISTANCE
                if (h.Position - op.Position).Magnitude > maxDist then
                    if now - (lastResyncTime[h] or 0) >= RESYNC_COOLDOWN then
                        local item = h.Parent
                        if item:IsA("Tool") and item.Parent == chr then
                            cleanupTool(h)
                            setupTool(item)
                        elseif item:IsA("Accessory") and item.Parent == chr then
                            cleanupTool(h)
                            item.Parent = nil
                            task.wait()
                            item.Parent = chr
                            task.wait()
                            setupAccessory(item) -- it used to get lost here, leaving only a few
                            local nh = item:FindFirstChild("Handle")
                            if nh then lastResyncTime[nh] = now end
                        end
                    end
                end
            end
        end
    end
end)
